📱 للـ Mobile App (مثال JavaScript/React Native):
═══════════════════════════════════════════════════════════════════

const loginUser = async (username, password) => {
  const formData = new URLSearchParams();
  formData.append('grant_type', 'password');
  formData.append('client_id', 'StoreManagement_Mobile');
  formData.append('username', username);
  formData.append('password', password);
  formData.append('scope', 'address email phone profile roles offline_access StoreManagement');

  try {
    const response = await fetch('https://localhost:44339/connect/token', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/x-www-form-urlencoded',
      },
      body: formData
    });

    const data = await response.json();
    
    // احفظ الـ tokens
    localStorage.setItem('access_token', data.access_token);
    localStorage.setItem('refresh_token', data.refresh_token);
    
    return data;
  } catch (error) {
    console.error('Login failed:', error);
    throw error;
  }
};

// استخدام الـ token في الطلبات
const getCategories = async () => {
  const token = localStorage.getItem('access_token');
  
  const response = await fetch('https://localhost:44339/api/app/category', {
    headers: {
      'Authorization': `Bearer ${token}`
    }
  });
  
  return response.json();
};

═══════════════════════════════════════════════════════════════════
⚠️ ملاحظات مهمة:
═══════════════════════════════════════════════════════════════════

1. الـ Content-Type يجب أن يكون: application/x-www-form-urlencoded
   ❌ ليس: application/json

2. الـ Body يجب أن يكون form-urlencoded
   ❌ ليس: JSON

3. الـ client_id هو: StoreManagement_Mobile
   ❌ ليس: StoreManagement_App

4. الـ scope يجب أن يشمل: offline_access للحصول على refresh_token
